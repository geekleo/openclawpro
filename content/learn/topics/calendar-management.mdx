## 前言

有效的日程管理是提升工作效率的关键。然而，日历工具往往是被动的——你需要主动打开日历查看，手动创建事件，自己记住准备事项。当会议密集或需要跨时区协调时，这些手动操作不仅耗时，还容易出错。

通过将 OpenClaw 与日历服务集成，你可以实现每日日程自动推送、会议智能准备、冲突实时检测、跨时区协调等自动化功能，让 AI 成为你的日程管理助手。

### 1.3 日历 API 客户端

```python
from google.oauth2.credentials import Credentials
from googleapiclient.discovery import build
from datetime import datetime, timedelta
import pytz

class CalendarClient:
    """Google Calendar API 客户端"""

    def __init__(self, credentials_path, token_path):
        creds = Credentials.from_authorized_user_file(token_path)
        self.service = build('calendar', 'v3', credentials=creds)
        self.timezone = pytz.timezone('Asia/Shanghai')

    def get_today_events(self):
        """获取今天的所有日程"""
        now = datetime.now(self.timezone)
        start = now.replace(hour=0, minute=0, second=0).isoformat()
        end = now.replace(hour=23, minute=59, second=59).isoformat()

        events_result = self.service.events().list(
            calendarId='primary',
            timeMin=start,
            timeMax=end,
            singleEvents=True,
            orderBy='startTime'
        ).execute()

        return events_result.get('items', [])

    def get_events_range(self, start_date, end_date):
        """获取指定日期范围的日程"""
        start = self.timezone.localize(
            datetime.combine(start_date, datetime.min.time())
        ).isoformat()
        end = self.timezone.localize(
            datetime.combine(end_date, datetime.max.time())
        ).isoformat()

        events_result = self.service.events().list(
            calendarId='primary',
            timeMin=start,
            timeMax=end,
            singleEvents=True,
            orderBy='startTime'
        ).execute()

        return events_result.get('items', [])

    def create_event(self, summary, start_time, end_time, description='',
                     location='', attendees=None, reminders=None):
        """创建日程事件"""
        event = {
            'summary': summary,
            'location': location,
            'description': description,
            'start': {
                'dateTime': start_time.isoformat(),
                'timeZone': 'Asia/Shanghai',
            },
            'end': {
                'dateTime': end_time.isoformat(),
                'timeZone': 'Asia/Shanghai',
            },
        }

        if attendees:
            event['attendees'] = [{'email': email} for email in attendees]

        if reminders:
            event['reminders'] = {
                'useDefault': False,
                'overrides': [
                    {'method': 'popup', 'minutes': m} for m in reminders
                ]
            }
        else:
            event['reminders'] = {
                'useDefault': False,
                'overrides': [
                    {'method': 'popup', 'minutes': 15},
                    {'method': 'popup', 'minutes': 5},
                ]
            }

        return self.service.events().insert(
            calendarId='primary', body=event
        ).execute()

    def check_conflicts(self, start_time, end_time, buffer_minutes=15):
        """检查时间段内是否有冲突"""
        # 扩大检查范围（加入缓冲时间）
        check_start = (start_time - timedelta(minutes=buffer_minutes)).isoformat()
        check_end = (end_time + timedelta(minutes=buffer_minutes)).isoformat()

        events = self.service.events().list(
            calendarId='primary',
            timeMin=check_start,
            timeMax=check_end,
            singleEvents=True,
            orderBy='startTime'
        ).execute()

        conflicts = events.get('items', [])
        return conflicts

    def find_free_slots(self, date, duration_minutes=60):
        """查找指定日期的空闲时段"""
        events = self.get_events_range(date, date)

        working_start = self.timezone.localize(
            datetime.combine(date, datetime.strptime("09:00", "%H:%M").time())
        )
        working_end = self.timezone.localize(
            datetime.combine(date, datetime.strptime("18:00", "%H:%M").time())
        )
        lunch_start = self.timezone.localize(
            datetime.combine(date, datetime.strptime("12:00", "%H:%M").time())
        )
        lunch_end = self.timezone.localize(
            datetime.combine(date, datetime.strptime("13:00", "%H:%M").time())
        )

        # 收集已占用时段
        busy_slots = [(lunch_start, lunch_end)]  # 午休
        for event in events:
            start = datetime.fromisoformat(event['start'].get('dateTime', ''))
            end = datetime.fromisoformat(event['end'].get('dateTime', ''))
            busy_slots.append((start, end))

        busy_slots.sort(key=lambda x: x[0])

        # 计算空闲时段
        free_slots = []
        current = working_start
        for busy_start, busy_end in busy_slots:
            if current + timedelta(minutes=duration_minutes) <= busy_start:
                free_slots.append((current, busy_start))
            current = max(current, busy_end)

        if current + timedelta(minutes=duration_minutes) <= working_end:
            free_slots.append((current, working_end))

        return free_slots
```

```python
async def create_event_from_text(calendar_client, openclaw_api, text):
    """从自然语言描述创建日程"""
    # 用 AI 解析自然语言
    parsed = await openclaw_api.chat(f"""
    请解析以下日程描述，返回 JSON 格式：
    {{
        "summary": "事件名称",
        "date": "YYYY-MM-DD",
        "start_time": "HH:MM",
        "end_time": "HH:MM",
        "location": "地点（如无则为空）",
        "description": "详细描述（如无则为空）",
        "attendees": ["参会人邮箱列表"],
        "reminders": [15, 5]
    }}

    用户输入：{text}
    今天日期：{datetime.now().strftime('%Y-%m-%d')}

    注意：
    - 如果用户说"明天"则日期为明天
    - 如果没有指定结束时间，默认持续 1 小时
    - 如果没有指定具体时间，根据事件类型智能推断
    """)

    import json
    event_data = json.loads(parsed)

    # 检查冲突
    start = datetime.strptime(
        f"{event_data['date']} {event_data['start_time']}", "%Y-%m-%d %H:%M"
    )
    end = datetime.strptime(
        f"{event_data['date']} {event_data['end_time']}", "%Y-%m-%d %H:%M"
    )

    tz = pytz.timezone('Asia/Shanghai')
    start = tz.localize(start)
    end = tz.localize(end)

    conflicts = calendar_client.check_conflicts(start, end)
    if conflicts:
        conflict_names = [c['summary'] for c in conflicts]
        return {
            "status": "conflict",
            "message": f"检测到日程冲突：{', '.join(conflict_names)}",
            "conflicts": conflicts,
            "suggested_event": event_data
        }

    # 创建事件
    result = calendar_client.create_event(
        summary=event_data['summary'],
        start_time=start,
        end_time=end,
        description=event_data.get('description', ''),
        location=event_data.get('location', ''),
        attendees=event_data.get('attendees'),
        reminders=event_data.get('reminders', [15, 5])
    )

    return {"status": "created", "event": result}
```

### 3.2 连续会议检测

```python
async def detect_meeting_marathon(calendar_client, openclaw_api, date):
    """检测连续会议并给出健康建议"""
    events = calendar_client.get_today_events()

    # 分析会议密度
    meeting_count = len(events)
    total_meeting_hours = sum(
        (datetime.fromisoformat(e['end']['dateTime']) -
         datetime.fromisoformat(e['start']['dateTime'])).seconds / 3600
        for e in events if 'dateTime' in e.get('start', {})
    )

    # 检测连续会议（间隔不足 15 分钟）
    consecutive_meetings = []
    for i in range(len(events) - 1):
        current_end = datetime.fromisoformat(events[i]['end'].get('dateTime', ''))
        next_start = datetime.fromisoformat(events[i + 1]['start'].get('dateTime', ''))
        gap = (next_start - current_end).total_seconds() / 60

        if gap < 15:
            consecutive_meetings.append({
                'event1': events[i]['summary'],
                'event2': events[i + 1]['summary'],
                'gap_minutes': gap
            })

    if meeting_count > 5 or total_meeting_hours > 5 or consecutive_meetings:
        advice = await openclaw_api.chat(f"""
        分析今天的日程健康度并给出建议：
        - 总会议数：{meeting_count}
        - 总会议时长：{total_meeting_hours:.1f} 小时
        - 连续会议（间隔<15分钟）：{len(consecutive_meetings)} 对

        请给出：
        1. 日程健康度评分（1-10）
        2. 具体建议（如何优化日程）
        3. 今天的深度工作时间预估
        """)
        return advice

    return None
```

## 五、跨时区协调

### 5.1 多时区空闲时段查找

```python
async def find_cross_timezone_slot(calendar_client, openclaw_api, participants, duration_minutes=60):
    """查找跨时区的共同空闲时段"""

    # 定义参与者和时区
    # participants = [
    #     {"name": "张明", "email": "zhang@company.com", "timezone": "Asia/Shanghai"},
    #     {"name": "John", "email": "john@company.com", "timezone": "America/New_York"},
    #     {"name": "Hans", "email": "hans@company.com", "timezone": "Europe/Berlin"},
    # ]

    # 使用 Google Calendar FreeBusy API
    body = {
        "timeMin": datetime.now(pytz.UTC).isoformat(),
        "timeMax": (datetime.now(pytz.UTC) + timedelta(days=7)).isoformat(),
        "timeZone": "UTC",
        "items": [{"id": p["email"]} for p in participants]
    }

    freebusy = calendar_client.service.freebusy().query(body=body).execute()

    # 解析忙碌时段
    busy_periods = {}
    for p in participants:
        calendar_data = freebusy.get("calendars", {}).get(p["email"], {})
        busy_periods[p["email"]] = calendar_data.get("busy", [])

    # 用 AI 分析最佳时间
    result = await openclaw_api.chat(f"""
    请找出以下参与者在未来 7 天内的共同空闲时段：

    参与者：
    {json.dumps(participants, ensure_ascii=False, indent=2)}

    忙碌时段（UTC）：
    {json.dumps(busy_periods, indent=2)}

    要求：
    1. 会议时长：{duration_minutes} 分钟
    2. 每个人的工作时间为当地时间 9:00-18:00
    3. 找出 3 个最佳选项
    4. 对每个选项，显示所有参与者的本地时间

    输出格式：
    选项 1：
      UTC: YYYY-MM-DD HH:MM - HH:MM
      张明 (上海): HH:MM - HH:MM
      John (纽约): HH:MM - HH:MM
      Hans (柏林): HH:MM - HH:MM
      推荐理由：...
    """)

    return result
```

```yaml
             - 如何优化会议安排以获得更多深度工作时间
          5. 下周日程预览（如有）

      - name: save_to_notion
        action: notion.create_page
        params:
          database_id: "${NOTION_REPORTS_DB}"
          properties:
            Name: "日程周报 - {{week_range}}"
            Type: "日程周报"
            Date: "{{today}}"
          content: "{{analysis}}"
```

## 总结

通过 OpenClaw 与日历服务的深度集成，你可以从被动的日程管理转变为主动的智能调度。每天早上自动收到日程简报，会议前自动准备上下文信息，冲突实时检测并建议替代方案，会后自动生成纪要——整个流程几乎不需要你手动干预。

关键要点：
1. 早间日程推送可以帮你快速了解一天的安排
2. 冲突检测 + 空闲时段查找可以大幅减少日程协调时间
3. 会议准备自动化可以让你以最佳状态进入每一场会议
4. 与 Notion 联动可以让会议纪要和日程回顾沉淀为知识资产
