## 前言

Discord 已经从最初的游戏语音工具发展为全球最受欢迎的社区平台之一，广泛应用于开源项目、Web3 社区、学习小组和技术团队中。其强大的 Bot 生态使得自动化运营成为可能，而将 OpenClaw AI Agent 接入 Discord 可以极大提升社区管理效率——从智能问答到自动化频道管理，从内容审核到定时推送。

本教程将完整讲解如何创建 Discord Bot、配置权限、开发交互功能，并与 OpenClaw 深度集成。

## 一、创建 Discord 应用与 Bot

### 1.1 创建应用

1. 访问 [Discord Developer Portal](https://discord.com/developers/applications)
2. 点击右上角「New Application」
3. 输入应用名称（如 `OpenClaw AI`），点击创建
4. 在应用设置页面，记录 **Application ID** 和 **Public Key**

### 1.2 创建 Bot 用户

1. 在左侧菜单中选择「Bot」
2. 点击「Add Bot」并确认
3. 设置 Bot 的用户名和头像
4. 点击「Reset Token」获取 **Bot Token**

**重要：Bot Token 是 Bot 的凭证，务必安全保管，不要泄露到公开仓库。**

```bash
# 将 Token 保存到环境变量
export DISCORD_BOT_TOKEN="your-bot-token-here"
```

### 1.3 邀请 Bot 到服务器

构建 OAuth2 邀请链接：

```
https://discord.com/api/oauth2/authorize?client_id=YOUR_APP_ID&permissions=8&scope=bot%20applications.commands
```

参数说明：
- `client_id`：替换为你的 Application ID
- `permissions=8`：管理员权限（生产环境建议使用最小权限原则）
- `scope`：`bot` 和 `applications.commands` 用于支持 Slash Commands

在浏览器中打开该链接，选择目标服务器，授权即可。

## 二、权限与 Intents 配置

### 2.1 Gateway Intents

Discord 使用 Intents 机制控制 Bot 可以接收的事件类型。需要在 Developer Portal 中启用特权 Intents：

1. 进入 Bot 设置页面
2. 在「Privileged Gateway Intents」区域启用：
   - **PRESENCE INTENT**：获取用户在线状态（用于显示活跃成员）
   - **SERVER MEMBERS INTENT**：获取成员加入/离开事件（用于欢迎消息）
   - **MESSAGE CONTENT INTENT**：读取消息内容（AI 对话必须开启）

### 2.2 权限配置

根据 Bot 功能需求，配置适当的权限位：

```javascript
// 推荐的 Bot 权限配置
const { PermissionFlagsBits } = require('discord.js');

const BOT_PERMISSIONS = [
  PermissionFlagsBits.ViewChannel,           // 查看频道
  PermissionFlagsBits.SendMessages,          // 发送消息
  PermissionFlagsBits.SendMessagesInThreads, // 在帖子中发送消息
  PermissionFlagsBits.EmbedLinks,            // 嵌入链接
  PermissionFlagsBits.AttachFiles,           // 附加文件
  PermissionFlagsBits.ReadMessageHistory,    // 读取历史消息
  PermissionFlagsBits.AddReactions,          // 添加反应
  PermissionFlagsBits.UseExternalEmojis,     // 使用外部表情
  PermissionFlagsBits.ManageMessages,        // 管理消息（删除、置顶）
  PermissionFlagsBits.ManageChannels,        // 管理频道
  PermissionFlagsBits.ManageRoles,           // 管理角色
  PermissionFlagsBits.CreatePublicThreads,   // 创建公开帖子
];
```

## 三、消息监听与回复

### 3.1 基础 Bot 框架

```javascript
// bot.js - Discord Bot 基础框架
const { Client, GatewayIntentBits, Partials } = require('discord.js');

const client = new Client({
  intents: [
    GatewayIntentBits.Guilds,
    GatewayIntentBits.GuildMessages,
    GatewayIntentBits.GuildMembers,
    GatewayIntentBits.MessageContent,
    GatewayIntentBits.DirectMessages,
  ],
  partials: [Partials.Channel, Partials.Message],
});

// Bot 就绪事件
client.once('ready', () => {
  console.log(`Bot 已上线: ${client.user.tag}`);
  console.log(`已加入 ${client.guilds.cache.size} 个服务器`);

  // 设置 Bot 状态
  client.user.setActivity('OpenClaw AI | /help', { type: 'WATCHING' });
});

// 消息监听与 AI 回复
client.on('messageCreate', async (message) => {
  // 忽略 Bot 自身的消息
  if (message.author.bot) return;

  // 检查是否在 AI 频道中
  const AI_CHANNELS = ['1234567890123456789', '1234567890123456790'];
  if (!AI_CHANNELS.includes(message.channel.id) && message.guild) return;

  try {
    // 显示「正在输入」状态
    await message.channel.sendTyping();

    // 调用 OpenClaw API 获取 AI 回复
    const response = await callOpenClawAPI(message.content, {
      userId: message.author.id,
      userName: message.author.username,
      channelId: message.channel.id,
      guildId: message.guild?.id,
    });

    // 处理长消息分段
    if (response.length > 1900) {
      const chunks = splitMessage(response, 1900);
      for (const chunk of chunks) {
        await message.reply(chunk);
      }
    } else {
      await message.reply(response);
    }
  } catch (error) {
    console.error('AI 回复错误:', error);
    await message.reply('抱歉，处理消息时出现了错误，请稍后重试。');
  }
});

// 消息分段函数
function splitMessage(text, maxLength) {
  const chunks = [];
  let remaining = text;
  while (remaining.length > 0) {
    if (remaining.length <= maxLength) {
      chunks.push(remaining);
      break;
    }
    // 尝试在换行处分割，保持内容完整性
    let splitIndex = remaining.lastIndexOf('\n', maxLength);
    if (splitIndex === -1 || splitIndex < maxLength * 0.5) {
      splitIndex = maxLength;
    }
    chunks.push(remaining.substring(0, splitIndex));
    remaining = remaining.substring(splitIndex).trimStart();
  }
  return chunks;
}

// 调用 OpenClaw API
async function callOpenClawAPI(content, context) {
  const response = await fetch('http://localhost:3000/api/chat', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/json',
      'Authorization': `Bearer ${process.env.OPENCLAW_API_KEY}`,
    },
    body: JSON.stringify({
      message: content,
      context: context,
      skill: 'discord',
    }),
  });
  const data = await response.json();
  return data.reply;
}

client.login(process.env.DISCORD_BOT_TOKEN);
```

// 监听消息反应，用于收藏、翻译等操作
client.on('messageReactionAdd', async (reaction, user) => {
  if (user.bot) return;

        `来自 #${originalMessage.channel.name} - ${originalMessage.author.username}\n` +
        `链接: ${originalMessage.url}`
      );

      console.log('无法发送私信给用户:', err);
    }
  }
});

### 4.2 处理 Slash Commands 交互

```javascript
// commands/handler.js - 处理 Slash Commands
client.on('interactionCreate', async (interaction) => {
  if (!interaction.isChatInputCommand()) return;

  const { commandName, options } = interaction;

  switch (commandName) {
    case 'ask': {
      // 延迟回复（AI 处理可能需要较长时间）
      await interaction.deferReply();

      const question = options.getString('question');
      const response = await callOpenClawAPI(question, {
        userId: interaction.user.id,
        channelId: interaction.channel.id,
      });

      await interaction.editReply(response);
      break;
    }

    case 'summarize': {
      await interaction.deferReply();
      const count = options.getInteger('count') || 20;

      // 获取频道历史消息
      const messages = await interaction.channel.messages.fetch({ limit: count });
      const messageTexts = messages
        .filter(m => !m.author.bot)
        .map(m => `${m.author.username}: ${m.content}`)
        .reverse()
        .join('\n');

      const summary = await callOpenClawAPI(
        `请用中文总结以下 Discord 频道对话的要点:\n${messageTexts}`,
        { userId: interaction.user.id }
      );

      await interaction.editReply(`**最近 ${count} 条消息的总结：**\n\n${summary}`);
      break;
    }

    case 'translate': {
      await interaction.deferReply();
      const text = options.getString('text');
      const target = options.getString('target');
      const langMap = { zh: '中文', en: '英文', ja: '日文', ko: '韩文' };

      const translated = await callOpenClawAPI(
        `请将以下文本翻译成${langMap[target]}，只输出翻译结果:\n${text}`,
        { userId: interaction.user.id }
      );

      await interaction.editReply(`**翻译结果（${langMap[target]}）：**\n${translated}`);
      break;
    }

    case 'reminder': {
      const reminderMsg = options.getString('message');
      const minutes = options.getInteger('minutes');

      await interaction.reply(`好的，我会在 ${minutes} 分钟后提醒你：${reminderMsg}`);

      setTimeout(async () => {
        await interaction.followUp(`<@${interaction.user.id}> 提醒你：${reminderMsg}`);
      }, minutes * 60 * 1000);
      break;
    }

    case 'help': {
      await interaction.reply({ embeds: [buildHelpEmbed()] });
      break;
    }
  }
});
```
