## 前言

Notion 是当今最流行的知识管理和团队协作工具之一，其灵活的数据库结构和丰富的内容类型使其成为个人笔记、项目管理、知识库构建的理想选择。然而，随着积累的内容越来越多，手动整理和维护 Notion 工作区变得越来越耗时。

通过将 OpenClaw 与 Notion API 集成，你可以实现笔记的自动创建、知识的智能分类、日报周报的模板化生成等功能，让 AI 成为你的知识管理助手。

## 一、Notion API 配置

### 1.1 创建 Notion Integration

1. 访问 [Notion Developers](https://www.notion.so/my-integrations)
2. 点击「New integration」
3. 填写信息：
   - Name：`OpenClaw Knowledge Manager`
   - Associated workspace：选择你的工作区
   - Type：Internal integration
4. 创建完成后，获取 **Internal Integration Secret**（以 `ntn_` 开头）

### 1.2 授权 Integration 访问页面

Notion Integration 默认没有任何页面的访问权限，你需要手动授权：

1. 打开你想让 OpenClaw 访问的 Notion 页面或数据库
2. 点击右上角「...」→「Add connections」
3. 搜索并选择你创建的 Integration（`OpenClaw Knowledge Manager`）
4. 确认授权

**注意：** 你需要为每一个需要访问的顶级页面/数据库单独授权。子页面会自动继承父页面的权限。

## 二、页面与数据库操作

### 2.1 Notion API 基础客户端

```python
import requests
import json
from datetime import datetime, timedelta

class NotionClient:
    """Notion API 客户端"""

    BASE_URL = "https://api.notion.com/v1"

    def __init__(self, api_key):
        self.api_key = api_key
        self.headers = {
            "Authorization": f"Bearer {api_key}",
            "Content-Type": "application/json",
            "Notion-Version": "2022-06-28"
        }

    # === 页面操作 ===

    def create_page(self, parent_id, title, content_blocks, properties=None, parent_type="database_id"):
        """在数据库或页面下创建新页面"""
        payload = {
            "parent": {parent_type: parent_id},
            "properties": properties or {},
            "children": content_blocks
        }
        if parent_type == "database_id" and "Name" not in payload["properties"]:
            payload["properties"]["Name"] = {
                "title": [{"text": {"content": title}}]
            }
        elif parent_type == "page_id":
            payload["properties"] = {
                "title": [{"text": {"content": title}}]
            }
        return requests.post(
            f"{self.BASE_URL}/pages",
            headers=self.headers,
            json=payload
        ).json()

    def get_page(self, page_id):
        """获取页面信息"""
        return requests.get(
            f"{self.BASE_URL}/pages/{page_id}",
            headers=self.headers
        ).json()

    def update_page_properties(self, page_id, properties):
        """更新页面属性"""
        return requests.patch(
            f"{self.BASE_URL}/pages/{page_id}",
            headers=self.headers,
            json={"properties": properties}
        ).json()

    def get_page_content(self, page_id):
        """获取页面内容（Block 列表）"""
        blocks = []
        url = f"{self.BASE_URL}/blocks/{page_id}/children"
        while url:
            resp = requests.get(url, headers=self.headers).json()
            blocks.extend(resp.get("results", []))
            next_cursor = resp.get("next_cursor")
            if next_cursor:
                url = f"{self.BASE_URL}/blocks/{page_id}/children?start_cursor={next_cursor}"
            else:
                url = None
        return blocks

    # === 数据库操作 ===

    def query_database(self, database_id, filter_obj=None, sorts=None, page_size=100):
        """查询数据库"""
        payload = {"page_size": page_size}
        if filter_obj:
            payload["filter"] = filter_obj
        if sorts:
            payload["sorts"] = sorts

        return requests.post(
            f"{self.BASE_URL}/databases/{database_id}/query",
            headers=self.headers,
            json=payload
        ).json()

    def create_database(self, parent_page_id, title, properties):
        """在页面下创建数据库"""
        payload = {
            "parent": {"page_id": parent_page_id},
            "title": [{"text": {"content": title}}],
            "properties": properties
        }
        return requests.post(
            f"{self.BASE_URL}/databases",
            headers=self.headers,
            json=payload
        ).json()

    # === 搜索 ===

    def search(self, query, filter_type=None, page_size=10):
        """搜索工作区内容"""
        payload = {"query": query, "page_size": page_size}
        if filter_type:
            payload["filter"] = {"property": "object", "value": filter_type}
        return requests.post(
            f"{self.BASE_URL}/search",
            headers=self.headers,
            json=payload
        ).json()
```

### 2.2 内容 Block 构建工具

```python
class NotionBlockBuilder:
    """Notion 内容块构建工具"""

    @staticmethod
    def heading(text, level=2):
        """创建标题块（level: 1/2/3）"""
        return {
            "object": "block",
            "type": f"heading_{level}",
            f"heading_{level}": {
                "rich_text": [{"type": "text", "text": {"content": text}}]
            }
        }

    @staticmethod
    def paragraph(text):
        """创建段落块"""
        return {
            "object": "block",
            "type": "paragraph",
            "paragraph": {
                "rich_text": [{"type": "text", "text": {"content": text}}]
            }
        }

    @staticmethod
    def bulleted_list(items):
        """创建无序列表"""
        return [
            {
                "object": "block",
                "type": "bulleted_list_item",
                "bulleted_list_item": {
                    "rich_text": [{"type": "text", "text": {"content": item}}]
                }
            }
            for item in items
        ]

    @staticmethod
    def numbered_list(items):
        """创建有序列表"""
        return [
            {
                "object": "block",
                "type": "numbered_list_item",
                "numbered_list_item": {
                    "rich_text": [{"type": "text", "text": {"content": item}}]
                }
            }
            for item in items
        ]

    @staticmethod
    def todo(text, checked=False):
        """创建待办事项"""
        return {
            "object": "block",
            "type": "to_do",
            "to_do": {
                "rich_text": [{"type": "text", "text": {"content": text}}],
                "checked": checked
            }
        }

    @staticmethod
    def code(code_text, language="python"):
        """创建代码块"""
        return {
            "object": "block",
            "type": "code",
            "code": {
                "rich_text": [{"type": "text", "text": {"content": code_text}}],
                "language": language
            }
        }

    @staticmethod
    def callout(text, emoji="bulb"):
        """创建高亮提示块"""
        return {
            "object": "block",
            "type": "callout",
            "callout": {
                "rich_text": [{"type": "text", "text": {"content": text}}],
                "icon": {"type": "emoji", "emoji": emoji}
            }
        }

    @staticmethod
    def divider():
        """创建分割线"""
        return {"object": "block", "type": "divider", "divider": {}}

    @staticmethod
    def toggle(text, children=None):
        """创建折叠块"""
        block = {
            "object": "block",
            "type": "toggle",
            "toggle": {
                "rich_text": [{"type": "text", "text": {"content": text}}],
            }
        }
        if children:
            block["toggle"]["children"] = children
        return block

    @staticmethod
    def table_of_contents():
        """创建目录块"""
        return {"object": "block", "type": "table_of_contents", "table_of_contents": {}}
```

## 三、自动知识整理工作流

### 3.1 智能分类与标签

```python
async def auto_classify_note(notion_client, openclaw_api, page_id):
    """使用 AI 自动为笔记分类和打标签"""
    # 获取页面内容
    blocks = notion_client.get_page_content(page_id)
    content_text = extract_text_from_blocks(blocks)

    # 调用 AI 进行分类
    classification = await openclaw_api.chat(f"""
    请分析以下笔记内容，返回 JSON 格式的分类结果：
    {{
        "category": "技术/产品/管理/生活/学习 之一",
        "tags": ["标签1", "标签2", "标签3"],
        "summary": "一句话摘要（不超过100字）",
        "quality": "Seed/Sapling/Evergreen"
    }}

    笔记内容：
    {content_text[:3000]}
    """)

    result = json.loads(classification)

    # 更新 Notion 页面属性
    notion_client.update_page_properties(page_id, {
        "Category": {"select": {"name": result["category"]}},
        "Tags": {"multi_select": [{"name": tag} for tag in result["tags"]]},
        "Summary": {"rich_text": [{"text": {"content": result["summary"]}}]},
        "Quality": {"select": {"name": result["quality"]}}
    })

    return result

def extract_text_from_blocks(blocks):
    """从 Notion blocks 中提取纯文本"""
    texts = []
    for block in blocks:
        block_type = block.get("type")
        if block_type and block_type in block:
            rich_text = block[block_type].get("rich_text", [])
            for rt in rich_text:
                texts.append(rt.get("plain_text", ""))
    return "\n".join(texts)
```

## 四、模板化创建

### 4.1 日报自动生成

```python
async def generate_daily_report(notion_client, openclaw_api, report_db_id, task_db_id):
    """自动生成每日工作报告"""
    today = datetime.now().strftime("%Y-%m-%d")
    tomorrow = (datetime.now() + timedelta(days=1)).strftime("%Y-%m-%d")

    # 查询今天完成的任务
    completed_tasks = notion_client.query_database(
        database_id=task_db_id,
        filter_obj={
            "and": [
                {"property": "Completed Date", "date": {"equals": today}},
                {"property": "Status", "status": {"equals": "Done"}}
            ]
        }
    )

    # 查询明天计划的任务
    planned_tasks = notion_client.query_database(
        database_id=task_db_id,
        filter_obj={
            "and": [
                {"property": "Due Date", "date": {"equals": tomorrow}},
                {"property": "Status", "status": {"does_not_equal": "Done"}}
            ]
        }
    )

    done_items = [get_page_title(t) for t in completed_tasks.get("results", [])]
    plan_items = [get_page_title(t) for t in planned_tasks.get("results", [])]

    # AI 生成日报内容
    report_content = await openclaw_api.chat(f"""
    请根据以下信息生成一份简洁的工作日报：

    今日完成（{len(done_items)} 项）：
    {chr(10).join(f'- {item}' for item in done_items) or '- 无'}

    明日计划（{len(plan_items)} 项）：
    {chr(10).join(f'- {item}' for item in plan_items) or '- 待规划'}

    请用简洁的中文输出，包含"今日工作总结"和"明日计划"两个部分，以及一句话的工作心得。
    """)

    # 构建页面内容
    bb = NotionBlockBuilder
    blocks = [
        bb.table_of_contents(),
        bb.heading(f"工作日报 - {today}", level=1),
        bb.callout(f"已完成 {len(done_items)} 项任务，明日计划 {len(plan_items)} 项", "clipboard"),
        bb.divider(),
        bb.heading("今日完成", level=2),
        *[bb.todo(item, checked=True) for item in done_items],
        bb.divider(),
        bb.heading("明日计划", level=2),
        *[bb.todo(item, checked=False) for item in plan_items],
        bb.divider(),
        bb.heading("AI 总结", level=2),
        bb.paragraph(report_content),
    ]

    # 创建日报页面
    properties = {
        "Name": {"title": [{"text": {"content": f"日报 - {today}"}}]},
        "Type": {"select": {"name": "日报"}},
        "Date": {"date": {"start": today}},
        "Status": {"status": {"name": "Draft"}}
    }

    result = notion_client.create_page(
        parent_id=report_db_id,
        title=f"日报 - {today}",
        content_blocks=blocks,
        properties=properties
    )
    return result


def get_page_title(page):
    """从 Notion 页面对象中提取标题"""
    title_prop = page.get("properties", {}).get("Name", {})
    title_list = title_prop.get("title", [])
    if title_list:
        return title_list[0].get("plain_text", "Untitled")
    return "Untitled"
```

```python
async def generate_weekly_report(notion_client, openclaw_api, report_db_id, task_db_id):
    """自动生成周报"""
    today = datetime.now()
    week_start = (today - timedelta(days=today.weekday())).strftime("%Y-%m-%d")
    week_end = today.strftime("%Y-%m-%d")

    # 查询本周完成的任务
    weekly_tasks = notion_client.query_database(
        database_id=task_db_id,
        filter_obj={
            "and": [
                {"property": "Completed Date", "date": {"on_or_after": week_start}},
                {"property": "Completed Date", "date": {"on_or_before": week_end}},
                {"property": "Status", "status": {"equals": "Done"}}
            ]
        }
    )

    done_items = [get_page_title(t) for t in weekly_tasks.get("results", [])]

    # 查询本周的日报
    daily_reports = notion_client.query_database(
        database_id=report_db_id,
        filter_obj={
            "and": [
                {"property": "Type", "select": {"equals": "日报"}},
                {"property": "Date", "date": {"on_or_after": week_start}},
                {"property": "Date", "date": {"on_or_before": week_end}}
            ]
        }
    )

    # AI 生成周报
    summary = await openclaw_api.chat(f"""
    请根据以下本周完成的工作，生成一份结构化的周报，包含：
    1. 本周成果概述（2-3句话）
    2. 关键工作详述（按重要性排序，每项 1-2 句描述）
    3. 遇到的问题与解决方案
    4. 下周工作计划（基于未完成任务推断）

    本周完成的 {len(done_items)} 项任务：
    {chr(10).join(f'- {item}' for item in done_items)}
    """)

    bb = NotionBlockBuilder
    blocks = [
        bb.table_of_contents(),
        bb.heading(f"周报 {week_start} ~ {week_end}", level=1),
        bb.callout(f"本周共完成 {len(done_items)} 项任务", "chart_increasing"),
        bb.divider(),
        bb.paragraph(summary),
        bb.divider(),
        bb.heading("任务完成清单", level=2),
        *[bb.todo(item, checked=True) for item in done_items],
    ]

    properties = {
        "Name": {"title": [{"text": {"content": f"周报 {week_start} ~ {week_end}"}}]},
        "Type": {"select": {"name": "周报"}},
        "Date": {"date": {"start": week_start, "end": week_end}},
        "Status": {"status": {"name": "Draft"}}
    }

    return notion_client.create_page(report_db_id, f"周报", blocks, properties)
```

## 常见问题

**Q: 提示 API Key 无效？**
确认已在目标页面/数据库中添加 Integration 连接（右上角 ... -> 添加连接）。每个顶级页面都需要单独授权。

**Q: 数据库属性类型报错？**
Notion API 对属性类型有严格要求，确保配置中的属性名称和类型与数据库中完全一致。使用 `get_page` 方法先获取页面信息来确认属性结构。

**Q: 创建页面时 children 超出限制？**
Notion API 单次请求最多添加 100 个 blocks。对于长内容，需要先创建页面再分批追加 blocks。

**Q: 内容格式不正确？**
Notion API 使用 Block 格式而非 Markdown。使用上文的 `NotionBlockBuilder` 工具类可以方便地构建各种 Block。

通过 OpenClaw 与 Notion 的深度集成，你可以将繁琐的知识管理工作交给 AI 自动处理。从快速笔记捕获到智能分类，从日报周报的自动生成到跨平台信息同步，每一个环节都可以实现自动化。
